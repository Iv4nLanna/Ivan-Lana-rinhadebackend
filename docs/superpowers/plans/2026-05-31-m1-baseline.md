```csharp
// src/Api/Data/ReferenceDataset.cs
using System.IO.Compression;
using System.Text.Json;
using System.Text.Json.Serialization;

namespace RinhaBackend.Data;

public class ReferenceDataset
{
    public float[,] Vectors { get; private set; } = new float[0, 14];
    public bool[] Labels { get; private set; } = [];
    public int Count { get; private set; }
    public Dictionary<string, float> MccRisk { get; private set; } = [];

    private volatile bool _isReady;
    public bool IsReady => _isReady;

    public async Task LoadAsync(string dataDir)
    {
        // Carrega mcc_risk.json
        var mccPath = Path.Combine(dataDir, "mcc_risk.json");
        var mccJson = await File.ReadAllTextAsync(mccPath);
        MccRisk = JsonSerializer.Deserialize<Dictionary<string, float>>(mccJson)
                  ?? throw new InvalidOperationException("mcc_risk.json inválido");

        // Carrega references.json.gz (ou example-references.json para testes)
        var fullPath = Path.Combine(dataDir, "references.json.gz");
        var examplePath = Path.Combine(dataDir, "example-references.json");

        ReferenceEntry[] entries;

        if (File.Exists(fullPath))
        {
            using var fileStream = File.OpenRead(fullPath);
            using var gzipStream = new GZipStream(fileStream, CompressionMode.Decompress);
            entries = await JsonSerializer.DeserializeAsync<ReferenceEntry[]>(gzipStream)
                      ?? throw new InvalidOperationException("references.json.gz inválido");
        }
        else if (File.Exists(examplePath))
        {
            var json = await File.ReadAllTextAsync(examplePath);
            entries = JsonSerializer.Deserialize<ReferenceEntry[]>(json)
                      ?? throw new InvalidOperationException("example-references.json inválido");
        }
        else
        {
            throw new FileNotFoundException(
                $"Nenhum arquivo de dataset encontrado em {dataDir}. " +
                "Esperado: references.json.gz ou example-references.json");
        }

        Count = entries.Length;
        Vectors = new float[Count, 14];
        Labels = new bool[Count];

        for (int i = 0; i < Count; i++)
        {
            for (int d = 0; d < 14; d++)
                Vectors[i, d] = entries[i].Vector[d];
            Labels[i] = entries[i].Label == "fraud";
        }

        _isReady = true;
    }

    // Tipo interno apenas para deserialização
    private sealed class ReferenceEntry
    {
        [JsonPropertyName("vector")]
        public float[] Vector { get; set; } = [];

        [JsonPropertyName("label")]
        public string Label { get; set; } = "";
    }
}
```

```csharp
// src/Api/Detection/Normalizer.cs
namespace RinhaBackend.Detection;

public static class Normalizer
{
    private static float Clamp(float x) => MathF.Max(0f, MathF.Min(1f, x));

    public static float[] Normalize(
        TransactionRequest req,
        Dictionary<string, float> mccRisk,
        HashSet<string> knownMerchants)
    {
        var vec = new float[14]; // M1: alocação no heap — será stackalloc em M4

        // 0: valor da transação normalizado (0–10000 → 0–1)
        vec[0] = Clamp((float)req.Transaction.Amount / 10_000f);

        // 1: número de parcelas (0–12 → 0–1)
        vec[1] = Clamp(req.Transaction.Installments / 12f);

        // 2: relação valor/média do cliente
        vec[2] = req.Customer.AvgAmount == 0m
            ? 0f
            : Clamp((float)(req.Transaction.Amount / req.Customer.AvgAmount) / 10f);

        // 3: hora do dia em UTC (0–23 → 0–1)
        vec[3] = req.Transaction.RequestedAt.Hour / 23f;

        // 4: dia da semana (Mon=0, Sun=6 → /6 → 0–1)
        //    .NET DayOfWeek: Sun=0, Mon=1..Sat=6 → transformar para Mon=0, Sun=6
        int dotNetDow = (int)req.Transaction.RequestedAt.DayOfWeek;
        int mondayBasedDow = (dotNetDow + 6) % 7;
        vec[4] = mondayBasedDow / 6f;

        // 5 e 6: dependem de last_transaction (null → -1)
        if (req.LastTransaction is null)
        {
            vec[5] = -1f;
            vec[6] = -1f;
        }
        else
        {
            // 5: minutos desde a última transação (0–1440min → 0–1)
            float minutes = (float)(req.Transaction.RequestedAt - req.LastTransaction.Timestamp).TotalMinutes;
            vec[5] = Clamp(minutes / 1_440f);

            // 6: km da última transação (0–1000km → 0–1)
            vec[6] = Clamp(req.LastTransaction.KmFromCurrent / 1_000f);
        }

        // 7: km de casa (0–1000km → 0–1)
        vec[7] = Clamp(req.Terminal.KmFromHome / 1_000f);

        // 8: número de transações nas últimas 24h (0–20 → 0–1)
        vec[8] = Clamp(req.Customer.TxCount24h / 20f);

        // 9: terminal online
        vec[9] = req.Terminal.IsOnline ? 1f : 0f;

        // 10: cartão presente
        vec[10] = req.Terminal.CardPresent ? 1f : 0f;

        // 11: merchant desconhecido para este cliente
        vec[11] = knownMerchants.Contains(req.Merchant.Id) ? 0f : 1f;

        // 12: risco da categoria MCC (default 0.5 se não encontrado)
        vec[12] = mccRisk.TryGetValue(req.Merchant.Mcc, out float risk) ? risk : 0.5f;

        // 13: ticket médio do merchant (0–10000 → 0–1)
        vec[13] = Clamp((float)req.Merchant.AvgAmount / 10_000f);

        return vec;
    }
}
```

- [ ] **Step 3: Implementar KnnSearch.cs**

> **Nota pedagógica:** este código aloca `new float[count]` por requisição propositalmente.
> Para 3M vetores isso são ~12 MB de heap por request. Em M2 vamos substituir
> por um tracker de top-5 que não aloca nada — e medir o impacto.

```csharp
// src/Api/Detection/KnnSearch.cs
namespace RinhaBackend.Detection;

public static class KnnSearch
{
    // M1: brute force ingênuo
    // Aloca float[count] para todas as distâncias — intencionalmente para ver o impacto em M2
    public static float Search(float[] query, float[,] vectors, bool[] labels, int count)
    {
        // ALOCAÇÃO M1: ~12 MB para 3M vetores — será eliminada em M2
        var distances = new float[count];

        // Calcula distância ao quadrado de cada vetor de referência até o query
        // (sem raiz quadrada — mantém a ordem para top-5 e é mais rápido)
        for (int i = 0; i < count; i++)
        {
            float sum = 0f;
            for (int d = 0; d < 14; d++)
            {
                float diff = query[d] - vectors[i, d];
                sum += diff * diff;
            }
            distances[i] = sum;
        }

        // Seleciona os 5 menores via LINQ (legível mas lento — será substituído em M2)
        var top5 = distances
            .Select((dist, idx) => (dist, idx))
            .OrderBy(x => x.dist)
            .Take(5)
            .ToArray();

        // Conta quantos dos 5 vizinhos são fraude
        int fraudCount = top5.Count(x => labels[x.idx]);
        return fraudCount / 5f;
    }
}
```
